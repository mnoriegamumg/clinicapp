@echo off
TITLE Iniciar Entorno de Desarrollo - ClinicApp
COLOR 0A

echo ========================================================
echo       INICIANDO ENTORNO DE DESARROLLO - CLINICAPP
echo ========================================================
echo.

:: 1. INICIAR MYSQL (Servicio de Windows o XAMPP)
echo [1/3] Iniciando servicio de MySQL...
net start MySQL >nul 2>&1
if %errorlevel% equ 0 (
    echo [OK] Servicio MySQL iniciado correctamente.
) else (
    :: Ruta alternativa común si usas XAMPP y el servicio no está registrado
    if exist "C:\xampp\mysql_start.bat" (
        start /min "MySQL XAMPP" C:\xampp\mysql_start.bat
        echo [OK] Servidor MySQL iniciado desde XAMPP.
    ) else (
        echo [AVISO] Asegurese de tener MySQL o XAMPP corriendo manualmente.
    )
)

echo.
:: 2. INICIAR BACKEND PHP
echo [2/3] Iniciando Backend (clinicapp-api)...
cd /d "%~dp0clinicapp-api"
if exist "artisan" (
    :: Si el proyecto utiliza Laravel / Artisan
    start "ClinicApp API" php artisan serve
) else (
    :: Servidor embebido genérico de PHP en puerto 8000
    start "ClinicApp API" php -S localhost:8000
)

echo.
:: 3. INICIAR FRONTEND ANGULAR
echo [3/3] Iniciando Frontend (clinicapp)...
cd /d "%~dp0clinicapp"
start "ClinicApp Frontend" npm start

echo.
echo ========================================================
echo       ¡ENTORNO DE CLINICAPP LEVANTADO CON ÉXITO!
echo ========================================================
echo - Backend API : http://localhost:8000 (o puerto configurado)
echo - Frontend    : http://localhost:4200
echo.
pause
